\section{Rappels: vecteurs sur le plan, 2D.}

\paragraph{Notations.}
\begin{itemize}
    \item Le plan en $2$-dimensions est un ensemble: $\R^2$.

    \item Notons un vecteur $\vec{v} \in \R^2$ 
    d'abscisse $x_{\vec{v}} \in \R$ et d'ordonnée $y_{\vec{v}} \in \R$: $$
        \vec{v} = \begin{pmatrix}
            x_{\vec{v}} \\
            y_{\vec{v}} \\
        \end{pmatrix}
    $$


    \item Notons, aussi, $\overrightarrow{AB}$ le vecteur décrivant le
    \enquote{mouvement} entre le point $A \in \R^2$ et le point $B \in \R^2$: $$
        \overrightarrow{AB} = \begin{pmatrix}
            x_B - x_A \\
            y_B - y_A \\
        \end{pmatrix}
    $$

    \item $
        \norm{\vec{v}}   
        = \sqrt{x_{\vec{v}}^2 + y_{\vec{v}}^2}
        \geq 0
    $: la norme, la longueur du vecteur $\vec{v}$.

    \item Vecteur \enquote{normalisé}, direction, de $\vec{v}$: $$
        \hat{u} = \frac{\vec{v}}{\norm{\vec{v}}},
        \quad \norm{\hat{u}} = 1
    $$ 

    \item $A \parallel B$: 
    \begin{itemize}
        \item Si $A$ et $B$ sont droites, segments, courbes:
            $A$ et $B$ sont \textit{parallèles}.
        \item Si $A$ et $B$ sont des vecteurs:
            $A$ et $B$ sont \textit{colinéaires}. 
    \end{itemize}
        
    \item $A \perp B$: 
    \begin{itemize}
        \item Si $A$ et $B$ sont droites, segments, courbes:
            $A$ et $B$ sont \textit{perpendiculaires}.
        \item Si $A$ et $B$ sont des vecteurs:
            $A$ et $B$ sont \textit{orthogonaux}.
    \end{itemize}
    
    \item $H$ est le \textit{projeté orthogonal} du point $A$
        sur $B$ (droite, plan) si: $$
            (AH) \perp B \text{ et } H = (AH) \cap B
        $$
    \item $\widehat{\vec{u}, \vec{v}}$: l'angle orienté entre les vecteurs
        $\vec{u}$ et $\vec{v}$.
\end{itemize}

\paragraph{Géométrie.}
\begin{itemize}
    \item Distance euclidienne: $$
        d = \sqrt{x^2 + y^2}
    $$
    \item Théorème de Thales: soit $2$ triangles $ABC$ et $ADE$: $$
        (BC) \parallel (DE) \iff
        \frac{AB}{AD} = \frac{AC}{AE} = \frac{BC}{DE}
    $$

    \item Milieu $I$ entre deux points $A$ et $B$: $$
        I = \frac{A + B}{2}
    $$

    \item Equation de la droite $(d)$ entre $2$ points $A$ et $B$ de pente $m$. $$
    \begin{aligned}
              &m = \frac{y_B - y_A}{x_B - x_A} \\
        (d):\;&y = m(x - x_A) + y_A = m(x - x_B) + y_B 
    \end{aligned}
    $$
    \item Equation du cercle $\mathcal{C}$ centré sur 
        le point $P$ de rayon $r \in \R$: $$
        \mathcal{C}: (x - x_P)^2 + (y - y_P)^2 = r^2
    $$
\end{itemize}


\paragraph{Vecteurs.}
\begin{itemize}
    \item Négation: $-\overrightarrow{AB} = \overrightarrow{BA}$
    \item Relation de Chasles (\enquote{joindre les bouts}): $
          \overrightarrow{AC} + \overrightarrow{CB}
        = \overrightarrow{AB}
    $
    \item Additions de vecteurs $\vec{u}$ et $\vec{v}$ dans $\R^2$: $$
      \begin{pmatrix} x_{\vec{u}} \\ y_{\vec{u}} \end{pmatrix}
    + \begin{pmatrix} x_{\vec{v}} \\ y_{\vec{v}} \end{pmatrix}
    = \begin{pmatrix} 
        x_{\vec{u}} + x_{\vec{v}} \\ 
        y_{\vec{u}} + y_{\vec{v}} \\ 
    \end{pmatrix} $$

    \item Facteur d'un vecteur $\vec{v} \in \R^2$ par $a \in \R$: $$
          a \times \begin{pmatrix} x_{\vec{v}} \\ y_{\vec{v}} \end{pmatrix}
        = \begin{pmatrix} a \times x_{\vec{v}} \\ a \times y_{\vec{v}} \end{pmatrix}
    $$
    \item Produit scalaire de vecteurs $\vec{u}$ et $\vec{v}$ dans $\R^2$: $$
    \begin{aligned}
        \vec{u} \cdot \vec{v} 
        & = x_{\vec{u}} \, x_{\vec{v}} + y_{\vec{u}} \, y_{\vec{v}} \\
        & = \norm{\vec{u}} \times \norm{\vec{v}} \times 
            \cos\of{\widehat{\vec{u}, \vec{v}}}
    \end{aligned}
    $$

\end{itemize}


\section{Géométrie dans un espace euclidien.}
En Terminale, nous étendons notre champ d'action: 
du plan, nous passons à l'espace 3-dimensionnel, dans $\R^3$.
Ainsi, à l'abscisse et à l'ordonné nous ajoutons la \enquote{hauteur}:
la \textit{cote} $z$ et notons, pour $\vec{v} \in \R^3$: $$
    \vec{v} = \begin{pmatrix}
        x_{\vec{v}} \\
        y_{\vec{v}} \\
        z_{\vec{v}} \\
    \end{pmatrix}
$$

Plus formellement, les vecteurs $\vec{i}, \vec{j}, \vec{k}$ 
forment une base $\mathcal{B}$ de $\R^3$ si, et seulement si: $$
    \forall \vec{t} \in \R^3, \quad
    \exists! (a, b, c) \in \R^3 \text{ tel que } 
    \vec{t} = a\vec{i} + b\vec{j} + c\vec{k} 
$$

\paragraph{Propriétés.}
Pour des vecteurs $\vec{u}, \vec{v}, \vec{w}$ dans $\R^3$.

\begin{itemize}
    \item Distance euclidienne dans $\R^3$, sur la même fondation que dans $\R^2$: $$
        d = \sqrt{x^2 + y^2 + z^2}
    $$
    \item $\vec{u} \parallel \vec{v}$: $\vec{u}$ et $\vec{v}$ sont \textit{colinéaires},
        comme dans $\R^2$, quand ils ont la même direction.
        Autrement dit: les droites $(u)$ et $(v)$ guidées par leur vecteur
        respectif sont parallèles $(u) \parallel (v)$. \\
        Formellement, disons: $$
            \exists! \lambda \in \R \quad \text{tel que} \quad
            \vec{u} = \lambda \vec{v}
        $$ 
    \item $\vec{u}, \vec{v}, \vec{w}$ sont \textit{coplanaires} s'ils sont compris
        dans une même base $\mathcal{B}$ de $\R^2$. \\
        Formellement, $\vec{u}$ peut s'écrire comme une 
        \textit{combinaison linéaire} de $\vec{v}$ et $\vec{w}$: $$
            \exists(\alpha, \beta) \in (\R^{\star})^2 \quad \text{tel que} \quad
            \vec{u} = \alpha \vec{v} + \beta \vec{w}
        $$
\end{itemize}


\paragraph{Figures.}
\begin{itemize}
    \item Représentation paramétrique de la droite $(d) \subset \R^3$,
        pour un point $P \in (d)$ et un vecteur directeur $\vec{v}$: $$
            (d) \; \begin{cases}
                x = x_P + x_{\vec{v}}\,t \\ 
                y = y_P + y_{\vec{v}}\,t \\ 
                z = z_P + z_{\vec{v}}\,t \\ 
            \end{cases}  \forall t \in \R
        $$
    \item Plan $\mathcal{P} \subset \R^3$: $$
        \mathcal{P}: ax + by + cz + d = 0
    $$
    \item Propriété des milieux, 
        $I$ milieu de $\left[AB\right], \forall M \in \R^3$: 
        $$
            2\overrightarrow{MI} = \overrightarrow{MA} + \overrightarrow{MB} 
        $$
    \item $ABCD$ parallélogramme $
        \iff \overrightarrow{AB} + \overrightarrow{AD} = \overrightarrow{AC}
        \iff \overrightarrow{AB} = \overrightarrow{DC}
    $
    \item Sphère $\mathcal{S}$ de rayon $r \in \R_{+}$ et de centre $C \in \R^3$: $$
        \mathcal{S}: (x - x_C)^2 + (y - y_C)^2 + (z - z_C)^2 = r^2
    $$
    
\end{itemize}


\paragraph{Quelques méthodes pour les exercices.}
En géométrie, il suffit souvent de remplacer les valeurs des équations
par les coordonnées des points, droites, plans que nous mettons en relation
et vérifier qu'il n'y aie pas de contradiction.
\begin{itemize}
    \item \textit{Si} un point $P$ appartient à une droite $(d)$, 
        \textit{Alors} $\exists! t \in \R$ tel que: $$
        P \in (d) \begin{cases}
            & x_P = a + xt \\ 
            & y_P = b + yt \\
            & z_P = c + zt \\
        \end{cases}
    $$ 
    \item Si 2 droites non parallèles $(d)$ et $(\Delta)$ s'intersectent en un point $P$,
        \textit{Alors} $\exists!(t, k) \in \R^2$ tel que: $$
        P = (d) \cap (\Delta) \begin{cases}
            & a + xt = a' + x'k \\
            & b + yt = b' + y'k \\
            & c + zt = c' + z'k \\
        \end{cases}
        $$
\end{itemize}   

\section{Orthogonalité.}
Il peut être très utile de chercher les rapport orthogonaux, 
étant la notions vectorielle équivalente de \enquote{parallèle}.

\paragraph{Produit scalaire.}
\begin{itemize}
    \item Vecteurs $u$ et $v$ orthogonaux $\iff \vec{u} \cdot \vec{v} = 0$. 
    \item Calculer le produit scalaire dans $\R^3$ (ressemble à $\R^2$): $$
    \begin{aligned}
        \vec{u} \cdot \vec{v} 
        & = \norm{u} \times \norm{v} \times \cos(\widehat{\vec{u}, \vec{v}}) \\
        & = x_{\vec{v}} \, x_{\vec{u}} 
          + y_{\vec{v}} \, y_{\vec{u}} 
          + z_{\vec{v}} \, z_{\vec{u}} \\
    \end{aligned} $$
    \item Carré scalaire de $\vec{v}$: $$
        \vec{v} \cdot \vec{v} = \vec{v}^2 = \norm{\vec{v}}^2
    $$

    \item 3 identités remarquables pour les vecteurs $u$ et $v$: 
        
        \begin{equation} \label{IR1}
            (\vec{u} + \vec{v})^2 = \vec{u}^2 + \vec{u} \cdot \vec{v} + \vec{v}^2
        \end{equation} \begin{equation} \label{IR2}
            (\vec{u} - \vec{v})^2 = \vec{u}^2 - \vec{u} \cdot \vec{v} + \vec{v}^2
        \end{equation} \begin{equation} \label{IR3}
            (\vec{u} + \vec{v}) \cdot (\vec{u} - \vec{v}) 
            = \vec{u}^2 + \vec{v}^2
        \end{equation}

    \item Formules de polarisation: $$
    \begin{aligned}
    \vec{u} \cdot \vec{v} 
    & = \frac{1}{2}\of{
        \norm{\vec{u} + \vec{v}}^2 
        - \norm{\vec{u}}^2 
        - \norm{\vec{v}}^2
    } \\
    & = \frac{1}{2}\of{
        \norm{\vec{u}}^2
        + \norm{\vec{v}}^2
        - \norm{\vec{u} + \vec{v}}^2
    } \\
    & = \frac{1}{4}\of{
        \norm{\vec{u} + \vec{v}}^2
        - \norm{\vec{u} - \vec{v}}^2
    }
    \end{aligned} $$
\end{itemize}

\paragraph{Droites et plans.}
Pour 2 droites dans $\R^2$, il y a 2 états possibles:
\textit{parallèles} et \textit{sécantes} (possiblement perpendiculaires). \\ 
Or, dans $\R^3$, un état s'ajoute: 
\textit{non sécantes et non parallèles}. \\

Dans $\R^3$:
\begin{itemize}
    \item Pour 2 droites $(d_1)$ et $(d_2)$, $(d_1) \perp (d_2)$ \textit{si},
        pour les vecteurs directeurs respectifs $\vec{d_1}$ et $\vec{d_2}$:
        $$
            \vec{d_1} \cdot \vec{d_2} = 0 
            \textbf{ et } (d_1) \cap (d_2)
        $$

    \item $\hat{n} = \vec{n}$ est le vecteur normal du plan $\mathcal{P}$ $$
    \begin{aligned}
        \iff & \hat{n} \perp \mathcal{P} \\        
        \iff & \forall \vec{t} \in \mathcal{P}: \hat{n} \cdot \vec{t} = 0 \\
        \iff & \vec{u}, \vec{v} \in \mathcal{P}, 
            \forall k \in \R \; \vec{u} \neq k\vec{v}: \\
        &\hat{n} \cdot \vec{u} = \hat{n} \cdot \vec{v} = 0 
    \end{aligned}
    $$
    
    Propriétés: $$
    \begin{aligned}
        & \norm{\hat{n}} = 1 \\
        \mathcal{P}: \: & x_{\hat{n}} + y_{\hat{n}} + z_{\hat{n}} + d = 0 \\
    \end{aligned}
    $$ 
    \item $H$ est le \textit{projeté orthogonal} de $A \in \R^3$ 
        sur le plan $\mathcal{P}$ \textit{si}, pour une droite $(d)$,
        guidée par le vecteur normal $\hat{n}$ à $\mathcal{P}$: $$
        \begin{aligned}
            H = (d) \cap \mathcal{P}
        \end{aligned} $$

    \item $\mathcal{P}$ est un \textit{plan médiateur} \textit{si},
        pour un segment $[AB] \subset \R^3$: $$
        \begin{aligned}
            & \forall M \in \mathcal{P} \\
            & AM = BM \\
            \iff & (x_M - x_A)^2 + (y_M - y_A)^2 + (z_M - z_A)^2 \\
               = & (x_M - x_B)^2 + (y_M - y_B)^2 + (z_M - z_B)^2
        \end{aligned} $$

\end{itemize}


\section{Pour aller plus loin.}

\paragraph{Produit croix.}
\begin{itemize}
    \item Pour $2$ vecteurs $\vec{u}$ et $\vec{v}$ dans $\R^2$: $$
        \vec{u} \times \vec{v} 
        = x_{\vec{u}} \, y_{\vec{v}} - y_{\vec{u}} \, x_{\vec{v}}
    $$ et peux représenter l'aire signé d'un parallélogramme formé 
    par $\vec{u}$ et $\vec{v}$.

    \item Pour $2$ vecteurs $\vec{u}$ et $\vec{v}$ dans $\R^n$, 
        plus généralement, nous avons: $$
        \vec{u} \times \vec{v} 
        = \norm{\vec{u}} \norm{\vec{v}} \sin\of{{\widehat{\vec{u}, \vec{v}}}}
    $$
    \item $\vec{u} \times \vec{v} = 0$: $\vec{u}$ et $\vec{v}$ sont colinéaires.
\end{itemize}

\paragraph{Généralisation pour $n$ dimensions.}
Nous pouvons aisément élargir quelques formules formées
comme des combinaisons linéaires et des additions.
Nous posons ici, pour vecteur (point ou vecteur géométrique) $\vec{x} \in \R^n$: $$
    \vec{x} = \begin{pmatrix}
        x_1 \\ x_2 \\ \dots \\ x_n
    \end{pmatrix}
$$

\begin{itemize}
    \item Distance euclidienne dans $R^n$: $$
        d = \sqrt{\sum_{i=1}^{n}{x_i^2}}
    $$
    \item Sphère $\mathcal{S}$ à $n$ dimensions 
        de centre $P \in \R^n$ et rayon $r$: $$
        \mathcal{S}: \sum_{i=1}^{n}{(x_i - P_i)^2} = r^2
    $$

    \item Pour qu'un vecteur $\overrightarrow{v_0} \in \R^n$ appartiennent à une base $\mathcal{B}$ 
        de $\R^{n - 1}$ définie par les vecteurs non colinéaires entre eux
        $\overrightarrow{v_1}, \overrightarrow{v_2}, \dots, \overrightarrow{v_{n-1}}$,
        doit être vérifiée cette combinaison linéaire: $$
            \exists \vec{\alpha} \in \R^{n-1} \text{ tel que }
            \overrightarrow{v_0} = \sum_{i=1}^{n-1}{\alpha_i \, \overrightarrow{v_i}}
        $$
\end{itemize}
